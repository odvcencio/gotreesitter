from pathlib import Path
import hashlib, json

p = Path('/evidence')
for name, digest in json.loads((p / 'fleet/input-pins.json').read_text()).items():
    assert hashlib.sha256((p / name).read_bytes()).hexdigest() == digest, name
for name, digest in json.loads((p / 'resume-input-pins.json').read_text()).items():
    assert hashlib.sha256((p / name).read_bytes()).hexdigest() == digest, name
assert (p / 'screen-pass.json').exists()
a = json.loads((p / 'rss-repeat-gomax1/audit.json').read_text())
b = json.loads((p / 'rss-repeat/audit.json').read_text())
assert a['environment']['GOMAXPROCS'] == '1'
assert a['binary_sha256'] == b['binary_sha256']
assert a['candidate_safety_pass'] and a['candidate_baseline_median_ratio'] <= 1.10
for role, digest in a['binary_sha256'].items():
    assert hashlib.sha256((p / f'fleet/{role}-rss-root.test').read_bytes()).hexdigest() == digest
    rows = [r for r in a['repeats'] if r['role'] == role]
    assert len(rows) == 20 and all(r['rss_bytes_per_source_byte'] <= 400 for r in rows)
assert len(a['initial']) == 6 and len(a['repeats']) == 40
initial_failures = [r for r in a['initial'] if r['rss_bytes_per_source_byte'] > 400]
assert len(initial_failures) == 1 and initial_failures[0]['role'] == 'baseline' and initial_failures[0]['case'] == 'go'
assert all(r['rss_bytes_per_source_byte'] <= 400 for r in a['initial'] if r['role'] == 'candidate')
(p / 'fleet/standalone-rss.json').write_text(json.dumps(a['initial'] + a['repeats'], indent=2) + '\n')
print('Retained initial baseline failure; candidate original/repeats and matched before/after RSS ceiling pass. Median ratchet passes.', flush=True)
