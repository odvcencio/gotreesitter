"""Run inside Docker with clean source checkouts mounted at /baseline and /workspace.

Arguments: published artifact directory, writable evidence directory.
Use Go 1.25.14 and the Docker settings in the receipt. This prepares the
focused checks and paired benchmarks; it never creates a passing gate marker.
"""
from pathlib import Path
import hashlib, json, shutil, sys

artifact, out = map(Path, sys.argv[1:3])
baseline = json.loads((artifact / 'frozen-source-pins.json').read_text())
source = json.loads((artifact / 'source-pins.json').read_text())
for name, digest in baseline['files'].items():
    assert hashlib.sha256((Path('/baseline') / name).read_bytes()).hexdigest() == digest, name
    if name not in source['candidate_files']:
        assert hashlib.sha256((Path('/workspace') / name).read_bytes()).hexdigest() == digest, name
for name, digest in source['candidate_files'].items():
    assert hashlib.sha256((Path('/workspace') / name).read_bytes()).hexdigest() == digest, name
out.mkdir(parents=True, exist_ok=True)
for src, dest in [('owned-benchmark.go.txt', 'owned-benchmark.go'),
                  ('owned-ledger.go.txt', 'owned-ledger.go'),
                  ('focused-checks.sh.txt', 'check.sh'), ('paired-perf.sh.txt', 'perf.sh')]:
    shutil.copyfile(artifact / 'reproduction' / src, out / dest)
replacements = {}
for root in ['/baseline', '/workspace']:
    replacements[root + '/benchmark_admission_candidate_route_test.go'] = str(out / 'owned-benchmark.go')
    replacements[root + '/cmd/perfcounterledger/main.go'] = str(out / 'owned-ledger.go')
(out / 'overlay.json').write_text(json.dumps({'Replace': replacements}, indent=2) + '\n')
print('Source pins verified. Run check.sh, then perf.sh under both benchmark locks.')
