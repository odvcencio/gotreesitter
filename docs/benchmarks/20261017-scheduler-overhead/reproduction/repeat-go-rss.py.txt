from pathlib import Path
from statistics import median
import hashlib, json, os, re, subprocess

p = Path('/evidence/fleet')
out = Path('/evidence/rss-repeat-gomax1')
out.mkdir(exist_ok=False)

def read_row(log, rss, role, case, stage, seed=None, position=None):
    text = log.read_text()
    assert '--- PASS:' in text and '--- SKIP:' not in text
    size = int(re.search(r'bytes=(\d+)', text)[1])
    peak = int(re.search(r'Maximum resident set size \(kbytes\): (\d+)', rss.read_text())[1]) * 1024
    if case == 'go':
        assert size == 1048591 and 'nodes=903086/' in text and 'tokens=425742 ' in text
        assert 'compact=1 fallback=0' in text and 'stopReason=accepted' in text
    return dict(role=role, case=case, stage=stage, seed=seed, position=position,
                input_bytes=size, peak_rss_bytes=peak, rss_bytes_per_source_byte=peak / size)

initial = []
for role in ['baseline', 'candidate']:
    for case in ['go', 'yaml_ordinary', 'yaml_owned']:
        initial.append(read_row(p / f'{role}-standalone-{case}.log',
                                p / f'{role}-standalone-{case}.rss', role, case, 'initial'))
rows = []
pins = {role: hashlib.sha256((p / f'{role}-rss-root.test').read_bytes()).hexdigest()
        for role in ['baseline', 'candidate']}
for seed in range(29101, 29121):
    order = ['baseline', 'candidate'] if seed % 2 else ['candidate', 'baseline']
    for position, role in enumerate(order, 1):
        log = out / f'{seed}-{role}.log'
        rss = out / f'{seed}-{role}.rss'
        args = ['/usr/bin/time', '-v', '-o', str(rss), str(p / f'{role}-rss-root.test'),
                '-test.run', '^TestCompactLineageRSS$/^go$', '-test.count=1',
                '-test.v', f'-test.shuffle={seed}', '-test.timeout=4m']
        with log.open('w') as stream:
            subprocess.run(args, cwd='/baseline' if role == 'baseline' else '/workspace',
                           check=True, stdout=stream, stderr=subprocess.STDOUT)
        row = read_row(log, rss, role, 'go', 'repeat', seed, position)
        rows.append(row)
        print(role, seed, round(row['rss_bytes_per_source_byte'], 3), flush=True)
summary = {}
for role in ['baseline', 'candidate']:
    values = [r['rss_bytes_per_source_byte'] for r in rows if r['role'] == role]
    summary[role] = dict(count=len(values), median=median(values), maximum=max(values),
                         ceiling_failures=sum(v > 400 for v in values))
candidate_failures = [r for r in initial + rows if r['role'] == 'candidate'
                      and r['rss_bytes_per_source_byte'] > 400]
ratio = summary['candidate']['median'] / summary['baseline']['median']
report = dict(protocol='20 alternating pairs of identical standalone Go safety probes; no timing claim',
              binary_sha256=pins,
              environment={k: os.environ.get(k) for k in ['GOGC', 'GODEBUG', 'GOMEMLIMIT', 'GOMAXPROCS', 'GOT_PARSE_NODE_LIMIT_SCALE', 'GOT_GLR_MAX_STACKS', 'GTS_ADMISSION_CANDIDATE', 'GTS_LAYOUT_OWNED_PROBE']},
              initial=initial, repeats=rows, summary=summary,
              candidate_ceiling_failures=candidate_failures, candidate_baseline_median_ratio=ratio,
              candidate_safety_pass=not candidate_failures and ratio <= 1.10,
              original_baseline_failure_retained=True)
(out / 'audit.json').write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps({k: v for k, v in report.items() if k not in ['initial', 'repeats']}, indent=2))
