//go:build gts_parsercorephase0 && gts_workcount

package gotreesitter

import (
	"encoding/json"
	"os"
	"testing"
)

func TestDeferredPublicationWorkCensus(t *testing.T) {
	for _, id := range []string{"real_sample", "rewrite", "query_compile", "language", "grammargen_lr"} {
		t.Run(id, func(t *testing.T) {
			var source []byte
			if id == "real_sample" {
				var err error
				source, err = os.ReadFile("internal/benchfixtures/testdata/real/go")
				if err != nil {
					t.Fatal(err)
				}
			} else {
				source = loadDiagnosticParserCoreCanonicalFixture(t, id).Source
			}
			p := newAdmissionCandidateGoParser(t)
			BeginDiagnosticWorkCount()
			tree, ok, reason := p.tryCompactFullParseRoute(source)
			parsed := EndDiagnosticWorkCount()
			if !ok || tree == nil {
				t.Fatalf("compact parse declined: %s", reason)
			}
			defer tree.Release()
			usedBefore := tree.arena.used
			BeginDiagnosticWorkCount()
			digest := requireDiagnosticParserCoreCanonicalTreeDigest(t, tree, p.language)
			consumed := EndDiagnosticWorkCount()
			if tree.RootNode().HasError() || tree.RootNode().EndByte() != uint32(len(source)) {
				t.Fatal("not a clean full-span tree")
			}
			data, err := json.Marshal(struct {
				Fixture        string
				DeepTree       string
				Parse          any
				Consume        any
				NodesBefore    int
				NodesAfter     int
				DeferredLeaves uint64
				FinalRefs      uint64
			}{id, digest, parsed, consumed, usedBefore, tree.arena.used, tree.arena.compactFullLeafCreated, tree.arena.finalChildRefsCreated})
			if err != nil {
				t.Fatal(err)
			}
			t.Logf("PUBLICATION_WORK_CENSUS %s", data)
		})
	}
}
