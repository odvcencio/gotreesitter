#!/usr/bin/env bash
set -euo pipefail
export PATH=/usr/local/go/bin:$PATH GOMAXPROCS=1 GOWORK=off GOFLAGS='-p=1 -mod=readonly -buildvcs=false'
export GTS_PARITY_C_REF_BUILD_CACHE=/workspace/harness_out/parity_c_ref_cache/linux_amd64
go version > /evidence/go-version.txt
cd /candidate
go test -c -tags gts_parsercorephase0 -o /evidence/root.test . > /evidence/root-build.log 2>&1
/evidence/root.test -test.run '^Test(TokenInvariant.*|GoTokenInvariant.*|IncrementalCheckpointReads.*|ProbeZeroWidthExternalToken.*)$' -test.count=1 -test.v -test.timeout=5m > /evidence/unit.log 2>&1
CGO_ENABLED=0 GOARCH=386 go test -c -tags gts_parsercorephase0 -o /evidence/root386.test . > /evidence/386-build.log 2>&1
/evidence/root386.test -test.run '^TestTokenInvariantPrimitive.*$' -test.count=1 -test.v -test.timeout=5m > /evidence/386-primitive.log 2>&1
for route in 0 1; do
 for lang in go yaml; do
  GTS_ADMISSION_CANDIDATE="$route" GOTREESITTER_V1_INVARIANT_LANGUAGE="$lang" /evidence/root.test -test.run '^TestV1InvariantGateR4EditSession$' -test.count=1 -test.v -test.timeout=5m > "/evidence/invariant-${route}-${lang}.log" 2>&1
 done
done
cd /candidate/cgo_harness
go test -c -tags treesitter_c_parity,gts_parsercorephase0 -overlay=/evidence/oracle-overlay.json -o /evidence/cgo.test . > /evidence/cgo-build.log 2>&1
/evidence/cgo.test -test.run '^TestGoTokenInvariantLookahead(NumericRepair|NumericReuse)LockedC$' -test.count=1 -test.v -test.timeout=5m > /evidence/numeric-C.log 2>&1
/evidence/cgo.test -test.run '^TestGoCognociente137KiBTopEditLockedC$/^[^/]+$/^[^/]+$/^(insert|replace)$' -test.count=1 -test.v -test.timeout=5m > /evidence/exact-go-C.log 2>&1
for route in 0 1; do
 for lang in go yaml python markdown; do
  GTS_ADMISSION_CANDIDATE="$route" GTS_PARITY_MODE=exhaustive /evidence/cgo.test -test.run "^(TestParityFreshParse|TestParityIncrementalParse|TestParityHasNoErrors)$/^${lang}$" -test.count=1 -test.v -test.timeout=5m > "/evidence/c-${route}-${lang}.log" 2>&1
 done
done
cd /candidate
go build -tags gts_parsercorephase0 -o /evidence/perfcounterledger ./cmd/perfcounterledger > /evidence/counter-build.log 2>&1
/evidence/perfcounterledger --write --output /evidence/selected-ledger-candidate.json > /evidence/counter.log 2>&1
printf 'complete\n' > /evidence/complete.txt
