#!/usr/bin/env bash
set -euo pipefail
export PATH=/usr/local/go/bin:$PATH GOMAXPROCS=1 GOWORK=off GOFLAGS='-p=1 -mod=readonly -buildvcs=false'
export GTS_PARITY_C_REF_BUILD_CACHE=/workspace/harness_out/parity_c_ref_cache/linux_amd64
go version > /evidence/go-version.txt
cd /candidate/cgo_harness
for route in 0 1; do
 while IFS= read -r lang; do
  case "$lang" in go|yaml|python|markdown) continue ;; esac
  GTS_ADMISSION_CANDIDATE="$route" GTS_PARITY_MODE=exhaustive /qualified/cgo.test -test.run "^(TestParityFreshParse|TestParityIncrementalParse|TestParityHasNoErrors)$/^${lang}$" -test.count=1 -test.v -test.timeout=5m > "/evidence/c-${route}-${lang}.log" 2>&1
 done < /evidence/languages.txt
done
for lang in djot toml; do
 /qualified/cgo.test -test.run "^TestIncrementalCReuseFreshRouteCorpus$/^${lang}$" -test.count=1 -test.v -test.timeout=5m > "/evidence/c-first-edit-${lang}.log" 2>&1
done
cd /candidate
for route in 0 1; do
 while IFS= read -r lang; do
  case "$lang" in go|yaml) continue ;; esac
  GTS_ADMISSION_CANDIDATE="$route" GOTREESITTER_V1_INVARIANT_LANGUAGE="$lang" /qualified/root.test -test.run '^TestV1InvariantGateR4EditSession$' -test.count=1 -test.v -test.timeout=5m > "/evidence/invariant-${route}-${lang}.log" 2>&1
 done < /evidence/languages.txt
done
/qualified/root.test -test.run '^Test(CompactAcceptance(ChangedSourceKeepsRebuildBudget|CachedPublicationStopControl)|CompactMemoryBudgetDetectsGrowthAfterSmallPoll|CompactAcceptance.*(MemoryBudget|RetentionBound)|CompactArenaReserve.*(Budget|RetentionCap)|CompactExternalScannerCheckpointCopyMemoryBudget|CompactRecoveryMemoMemoryBudget|CompactBorrowedMaterializationProjectionMemoryBudget|G18RefSetSchedulerMemoryBudgetAccounting|RuntimeMemoryBudget.*|RuntimeMemoryHardCeilingStillStopsParse|ParseMemoryHardCeiling.*)$' -test.count=1 -test.v -test.timeout=8m > /evidence/memory-contracts.log 2>&1
go test -race -tags gts_parsercorephase0 -run '^Test(TokenInvariantPrimitive.*|TokenInvariantExternal.*|IncrementalCheckpointReads.*|ProbeZeroWidthExternalToken.*)$' -count=1 -v . > /evidence/race.log 2>&1
printf 'complete\n' > /evidence/complete.txt
