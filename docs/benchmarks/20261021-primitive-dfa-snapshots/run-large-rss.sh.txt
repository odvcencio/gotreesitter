#!/usr/bin/env bash
set -euo pipefail
export PATH=/evidence/bin:/usr/local/go/bin:$PATH GOMAXPROCS=1 GOWORK=off GOFLAGS='-p=1 -mod=readonly -buildvcs=false -overlay=/evidence/overlay.json'
go version > /evidence/go-version.txt
for variant in baseline candidate; do
 GTS_FRESH_MEMORY_LANG=go GTS_ADMISSION_CANDIDATE=0 "/previous/${variant}-bench.test" -test.run '^$' -test.bench '^BenchmarkFreshMemory$/^go$/^1024KiB$' -test.benchtime=1x -test.count=1 -test.benchmem -test.timeout=5m > "/evidence/${variant}-preflight.log" 2>&1
 awk '/^Benchmark/ {for(i=4;i<=NF;i+=2) m[$i]=$(i-1);print m["tokens/op"],m["nodes/op"],m["stacks/op"],m["compact-routes/op"],m["compact-fallbacks/op"],m["input-B/op"],m["arena-B/input-B"]}' "/evidence/${variant}-preflight.log" > "/evidence/${variant}-preflight-counters.txt"
done
test -s /evidence/baseline-preflight-counters.txt
cmp /evidence/baseline-preflight-counters.txt /evidence/candidate-preflight-counters.txt
cd /candidate
GTS_FRESH_MEMORY_LANG=go GTS_ADMISSION_CANDIDATE=0 bash scripts/run_randomized_benchmarks.sh --output /evidence/go-after.txt --baseline-root /baseline --baseline-output /evidence/go-before.txt --lock-path /evidence/benchmark.lock --package ./roottest/bench --bench-regex '^BenchmarkFreshMemory$/^go$/^1024KiB$' --require-benchmarks BenchmarkFreshMemory/go/1024KiB --runs 20 --seed-start 136001 --benchtime 750ms > /evidence/go-runner.log 2>&1
printf 'complete\n' > /evidence/complete.txt
