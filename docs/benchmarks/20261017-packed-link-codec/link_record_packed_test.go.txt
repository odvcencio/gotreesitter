package parsercorephase0

import (
	"math"
	"math/rand"
	"testing"
	"unsafe"
)

func TestPackedLinkRecordBoundaryRoundTrips(t *testing.T) {
	if size := unsafe.Sizeof(packedLinkRecord{}); size != 16 {
		t.Fatalf("hot link size=%d", size)
	}
	for _, row := range []struct {
		name   string
		score  int64
		order  uint64
		flags  uint32
		inline bool
	}{
		{"zero", 0, 0, 0, true},
		{"known_zero_order", 0, 0, linkFlagHasOrder, true},
		{"unflagged_order", 0, 17, 0, true},
		{"zero_score_largest_order", 0, (1 << 30) - 1, linkFlagHasOrder, true},
		{"negative_score", -1, 0, 0, true},
		{"small_min", math.MinInt16, (1 << 13) - 1, linkFlagHasOrder, true},
		{"small_max", math.MaxInt16, (1 << 13) - 1, 0, true},
		{"small_order_overflow", -1, 1 << 13, linkFlagHasOrder, false},
		{"positive_score_overflow", math.MaxInt16 + 1, 0, 0, false},
		{"negative_score_overflow", math.MinInt16 - 1, 0, 0, false},
		{"zero_score_order_overflow", 0, 1 << 30, linkFlagHasOrder, false},
		{"full_signed_min", math.MinInt64, 0, 0, false},
		{"full_signed_max", math.MaxInt64, 0, 0, false},
		{"full_unsigned_order", 0, math.MaxUint64, linkFlagHasOrder, false},
		{"recovery", 0, 0, linkFlagRecoveryDiscontinuity, false},
		{"unknown_flags", 0, 0, 1 << 31, false},
	} {
		t.Run(row.name, func(t *testing.T) {
			original := linkRecord{scoreDelta: row.score, order: row.order, flags: row.flags, prev: NodeID(math.MaxUint32), payload: SubtreeID(math.MaxUint32), next: LinkID(math.MaxUint32)}
			packed, ok := packInlineLinkRecord(original)
			if ok != row.inline {
				t.Fatalf("inline=%t want=%t", ok, row.inline)
			}
			got, decoded := packed.unpackInline()
			if decoded != ok || ok && got != original {
				t.Fatalf("metadata/reference corruption: original=%+v decoded=%+v/%t", original, got, decoded)
			}
			if !ok && packed.metadata != packedLinkColdMetadata {
				t.Fatal("wide metadata did not request complete-width storage")
			}
		})
	}
}

func TestPackedLinkRecordGeneratedRoundTrips(t *testing.T) {
	random := rand.New(rand.NewSource(450196))
	for i := 0; i < 10000; i++ {
		original := linkRecord{prev: NodeID(random.Uint32()), payload: SubtreeID(random.Uint32()), next: LinkID(random.Uint32()), flags: random.Uint32() & linkFlagHasOrder}
		switch i % 3 {
		case 0:
			original.order = random.Uint64() & ((1 << 30) - 1)
		case 1:
			original.scoreDelta = int64(int16(random.Uint32()))
			original.order = random.Uint64() & ((1 << 13) - 1)
		case 2:
			original.scoreDelta = int64(random.Uint64())
			original.order = random.Uint64()
			original.flags = random.Uint32()
		}
		packed, ok := packInlineLinkRecord(original)
		decoded, unpacked := packed.unpackInline()
		if ok != unpacked || ok && original != decoded {
			t.Fatalf("case %d changed: %+v -> %+v", i, original, decoded)
		}
		if i%3 != 2 && !ok {
			t.Fatalf("representable metadata escaped inline storage at %d", i)
		}
	}
}
